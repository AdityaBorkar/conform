## Engine

### `check(targetPath: string, opts?: ConformanceOptions): Promise<ConformanceResult | MonorepoConformanceResult | ConformanceError>`

`src/api/engine.ts:266` — auto-detects monorepo via `loadRawConfig` + `isMonorepoConfig` → delegates to `checkMonorepo` or single-repo flow.

```ts
const result = await check("/path/to/repo", { verbose: true, json: false, groupBy: "domains" });

if ("code" in result) {
  // ConformanceError
  console.error(result.code, result.message);
} else if ("packageResults" in result) {
  // MonorepoConformanceResult
} else {
  // ConformanceResult
}
```

### `checkMonorepo(rootDir: string, opts?: ConformanceOptions): Promise<MonorepoConformanceResult | ConformanceError>`

`src/api/engine.ts:309` — `loadAndResolveMonorepo` → per-package `buildPackageConformance` → aggregated `MonorepoConformanceResult` (or `renderJson` envelope when `json: true`).

### `runChecks(preset: Preset, targetPath: string): Promise<RuleResult[]>`

`src/api/engine.ts:172` — core loop: flatten `preset.plugins[].rules`, `parseOverride`, unknown `level` → `fail`, `off` → skip, `rule.check(targetPath, params)`, `coerceStatus`.

## Types

`src/types.ts`

```ts
type Severity = "pass" | "warn" | "fail";
type GroupBy  = "domains" | "files";
type RuleLevel = "warn" | "off" | "error";

interface CheckResult { status: Severity; message?: string; }

interface Rule<P = unknown> {
  id: string; domain: string; files: string[]; description: string;
  check: (ctx: string, params?: P) => CheckResult | Promise<CheckResult>;
  paramsSchema?: Type<P>;
}

interface Plugin<Id = string, ParamMap = Record<string, unknown>> {
  _paramMap: ParamMap; id: Id; rules: Rule<unknown>[];
}

type RuleConfig<P> = RuleLevel | ({ level?: RuleLevel } & Partial<P>);
type RuleOverrides = Record<string, RuleConfig<unknown>>;
type StrictRuleOverrides = { [K in keyof RuleRegistry]?: RuleConfig<RuleRegistry[K]> } & Record<string, RuleConfig<unknown>>;
type StrictPresetRules<Ps> = { [K in keyof PresetRuleRegistry<Ps>]?: RuleConfig<PresetRuleRegistry<Ps>[K]> } & Record<string, RuleConfig<unknown>>;
type RuleRegistry = InferPresetParamMap<AllBuiltinPlugins>; // auto-inferred from barrel

interface Preset { name: string; description: string; plugins: Plugin[]; rules?: StrictRuleOverrides; }
interface PresetWithPlugins<Ps> { name: string; description: string; plugins: Ps; rules?: StrictPresetRules<Ps>; }

interface RuleResult { id: string; domain: string; files: string[]; description: string; status: Severity; message?: string; }
interface ConformConfig { preset: string; plugins?: Plugin[]; rules?: StrictRuleOverrides; }
type MonorepoConfig = Record<string, ConformConfig>;
interface ConformOutput { preset: string; path: string; results: RuleResult[]; summary: { pass: number; warn: number; fail: number }; groupBy?: GroupBy; }
interface MonorepoConformOutput { path: string; outputs: ConformOutput[]; summary: { pass: number; warn: number; fail: number }; }
interface MonorepoPackageResult { path: string; output: ConformOutput; rendered: string; hasFail: boolean; hasWarn: boolean; }

interface PackageJson { name?, version?, license?, type?, description?, engines?, homepage?, repository?, sideEffects?, bin?, bugs?, contributors?, dependencies?, devDependencies?, peerDependencies?, exports?, files?, main?, module?, scripts?, workspaces?, author? }
```
